python -m nuitka --onefile --windows-console-mode=disable --enable-plugin=pyqt6 --include-package=requests --include-package=websocket --include-package=cryptography --include-package=fpdf --include-data-file=data/DejaVuSans.ttf=data/DejaVuSans.ttf --windows-icon-from-ico=icon.ico --assume-yes-for-downloads --output-dir=build --output-filename=Legalyze.exe main.py

REM ============================================================================
REM Legalyze — сборка ОДНОГО .exe файла (Nuitka onefile).
REM Запуск команды из папки клиента (где лежат main.py, icon.ico и data/DejaVuSans.ttf) на Windows.
REM Результат сборки: build\Legalyze.exe (единый исполняемый файл с иконкой и без консоли).
REM ============================================================================
REM Подготовка окружения перед первой сборкой:
REM   pip install --upgrade pip
REM   pip install nuitka pyqt6 requests websocket-client cryptography fpdf2
REM
REM Разбор параметров команды:
REM   --onefile                       — упаковать проект и все зависимости в 1 исполняемый .exe файл
REM   --windows-console-mode=disable  — отключить появление консольного чёрного окна при запуске
REM   --enable-plugin=pyqt6           — включить плагин поддержки графического фреймворка PyQt6
REM   --include-package=requests      — включить библиотеку HTTP-запросов к серверу
REM   --include-package=websocket     — включить websocket-клиент для управления браузером по CDP
REM   --include-package=cryptography  — включить библиотеку шифрования Fernet / AES
REM   --include-package=fpdf          — включить генератор PDF для legacy-формата промтов
REM   --include-data-file=data/DejaVuSans.ttf=data/DejaVuSans.ttf — встроить шрифт DejaVu Sans
REM   --windows-icon-from-ico=icon.ico — встроить иконку icon.ico в исполняемый файл .exe
REM   --assume-yes-for-downloads      — автоматически подтверждать загрузку компилятора C (MinGW64)
REM   --output-dir=build              — папка для сохранения результата сборки
REM   --output-filename=Legalyze.exe  — имя итогового исполняемого файла
REM
REM Рядом с полученным Legalyze.exe обязательно должна располагаться папка «chromium» с chrome.exe.
